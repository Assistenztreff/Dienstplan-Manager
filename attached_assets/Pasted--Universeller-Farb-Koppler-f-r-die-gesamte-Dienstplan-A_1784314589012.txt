/**
 * Universeller Farb-Koppler für die gesamte Dienstplan-App.
 * Nimmt einen Farb- oder Status-Key entgegen und gibt ein barrierefreies (WCAG AA)
 * Set aus Hintergrund-, Text- und Rahmen-Klassen zurück[cite: 1].
 */
export function getBarrierefreieFarbe(keyString: string): string {
  if (!keyString) return "bg-gray-100 text-assistenz-brand border-gray-300";
  
  const key = keyString.toLowerCase().trim();

  switch (key) {
    // Status: Aktiv, Standard-Zellen, Assistenzkraft Typ 1
    case "mint":
    case "hellblau":
    case "active":
      return "bg-assistenz-mint text-assistenz-brand border-assistenz-brand/20";
      
    // Status: Entwurf, Offen, Warnung, Assistenzkraft Typ 2
    case "yellow":
    case "hellgelb":
    case "pending":
    case "draft":
      return "bg-assistenz-yellow text-assistenz-brand border-assistenz-brand/20";
      
    // Status: Bestätigt, Fixiert, Abgeschlossen, Assistenzkraft Typ 3
    case "green":
    case "grün":
    case "confirmed":
    case "success":
      return "bg-assistenz-green text-assistenz-brand border-assistenz-brand/20";
      
    // Status: Pause, Wichtige Info, System-Meldung, Assistenzkraft Typ 4
    case "peach":
    case "pfirsich":
    case "break":
    case "info":
      return "bg-assistenz-peach text-assistenz-darkText border-assistenz-darkText/20";
      
    // Erweiterte Assistenz- / Kategorie-Farben (Optionen 5 bis 8)
    case "purple":
    case "lila":
      return "bg-assistenz-purple text-assistenz-darkText border-assistenz-darkText/20";
    case "pink":
    case "rosa":
      return "bg-assistenz-pink text-assistenz-brand border-assistenz-brand/20";
    case "terracotta":
    case "orange":
      return "bg-assistenz-terracotta text-assistenz-darkText border-assistenz-darkText/20";
    case "sage":
    case "salbei":
      return "bg-assistenz-sage text-assistenz-brand border-assistenz-brand/20";
      
    default:
      // Absolut sicherer, kontraststarker System-Fallback
      return "bg-gray-100 text-assistenz-brand border-gray-200";
  }
}
